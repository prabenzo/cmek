```sql
CREATE TABLE deks (id TEXT PRIMARY KEY, tenant_id TEXT NOT NULL, kek_id TEXT NOT NULL, kek_version INTEGER NOT NULL,
  wrapped_dek BLOB NOT NULL, created_at INTEGER NOT NULL);
CREATE TABLE messages (
  id INTEGER PRIMARY KEY, tenant_id TEXT NOT NULL, dek_id TEXT NOT NULL, nonce BLOB NOT NULL, ciphertext BLOB NOT NULL,
  state TEXT NOT NULL CHECK (state IN ('ready','claimed','dead')), attempts INTEGER NOT NULL DEFAULT 0,
  enqueued_at INTEGER NOT NULL, claimed_until INTEGER);
CREATE INDEX messages_tenant_state_id ON messages (tenant_id, state, id);
CREATE INDEX messages_claimed_until   ON messages (claimed_until) WHERE state = 'claimed';
-- Insert   (ledger: accepted[i]++ ready[i]++ total++; backlogged++ if backlog was 0; wake; insert timing)
INSERT INTO messages (id, tenant_id, dek_id, nonce, ciphertext, state, attempts, enqueued_at) VALUES (?1,?2,?3,?4,?5,'ready',0,?6);
-- Claim    (cursor drained fully ignoring ctx; n = rows drained, SELECT changes() on scan error [SC-F10]; ready[i] -= n; claimed[i] += n)
UPDATE messages SET state='claimed', claimed_until=?1
  WHERE id IN (SELECT id FROM messages WHERE tenant_id=?2 AND state='ready' ORDER BY id LIMIT ?3)
  RETURNING id, dek_id, nonce, ciphertext, attempts, enqueued_at;
-- Ack      (ra = rowsAffected; delivered[i] += ra; claimed[i] -= ra; total -= ra; backlogged-- if backlog hits 0)
DELETE FROM messages WHERE state='claimed' AND id IN (SELECT value FROM json_each(?1));
-- Release  (claimed[i] -= ra; ready[i] += ra; wake)
UPDATE messages SET state='ready', claimed_until=NULL WHERE state='claimed' AND id IN (SELECT value FROM json_each(?1));
-- Dead     (claimed[i] -= ra; dead[i] += ra)
UPDATE messages SET state='dead', claimed_until=NULL, attempts=attempts+1 WHERE id=?1 AND state='claimed';
```

- **The mean threshold may be crossed by checkpoints alone.** One auto-checkpoint of `s` ms per second adds ≈ 0.5·s² µs to the mean at 1,500/s (10 ms → +50 µs, 20 ms → +200 µs on top of the 150–300 µs per-insert cost), and on a tmpfs neither `synchronous=OFF` nor a larger autocheckpoint removes it. - **Unbounded backlog at 1,500/s (no admission until M3): ≈ 51,600 rows, WAL churn.** Mitigation: 60 s only; `wal_autocheckpoint=4000` pre-applied; `Claim`'s subquery uses `messages_tenant_state_id`; the run is killed before the 300/s check.
